         case 4 ; Set axis impedance (per-axis stiffness/damping)
               gAxisImpParamsValid = FALSE

               gAxisImpParams.STIFFNESS.A1 = 0.0
               gAxisImpParams.DAMPING.A1 = 0.0
               gAxisImpParams.STIFFNESS.A2 = 0.0
               gAxisImpParams.DAMPING.A2 = 0.0
               gAxisImpParams.STIFFNESS.A3 = 0.0
               gAxisImpParams.DAMPING.A3 = 0.0
               gAxisImpParams.STIFFNESS.A4 = 0.0
               gAxisImpParams.DAMPING.A4 = 0.0
               gAxisImpParams.STIFFNESS.A5 = 0.0
               gAxisImpParams.DAMPING.A5 = 0.0
               gAxisImpParams.STIFFNESS.A6 = 0.0
               gAxisImpParams.DAMPING.A6 = 0.0

               IF gMxA_CmdQ_Int1[CommandQueueIndex] >= 2 THEN
                  gAxisImpParams.STIFFNESS.A1 = gMxA_CmdQ_Real1[CommandQueueIndex]
                  gAxisImpParams.DAMPING.A1 = gMxA_CmdQ_Real2[CommandQueueIndex]
               ENDIF
               IF gMxA_CmdQ_Int1[CommandQueueIndex] >= 4 THEN
                  gAxisImpParams.STIFFNESS.A2 = gMxA_CmdQ_Real3[CommandQueueIndex]
                  gAxisImpParams.DAMPING.A2 = gMxA_CmdQ_Real4[CommandQueueIndex]
               ENDIF
               IF gMxA_CmdQ_Int1[CommandQueueIndex] >= 6 THEN
                  gAxisImpParams.STIFFNESS.A3 = gMxA_CmdQ_Real5[CommandQueueIndex]
                  gAxisImpParams.DAMPING.A3 = gMxA_CmdQ_Real6[CommandQueueIndex]
               ENDIF
               IF gMxA_CmdQ_Int1[CommandQueueIndex] >= 8 THEN
                  gAxisImpParams.STIFFNESS.A4 = gMxA_CmdQ_Real7[CommandQueueIndex]
                  gAxisImpParams.DAMPING.A4 = gMxA_CmdQ_Real8[CommandQueueIndex]
               ENDIF
               IF gMxA_CmdQ_Int1[CommandQueueIndex] >= 10 THEN
                  gAxisImpParams.STIFFNESS.A5 = gMxA_CmdQ_Real9[CommandQueueIndex]
                  gAxisImpParams.DAMPING.A5 = gMxA_CmdQ_Real10[CommandQueueIndex]
               ENDIF
               IF gMxA_CmdQ_Int1[CommandQueueIndex] >= 12 THEN
                  gAxisImpParams.STIFFNESS.A6 = gMxA_CmdQ_Real11[CommandQueueIndex]
                  gAxisImpParams.DAMPING.A6 = gMxA_CmdQ_Real12[CommandQueueIndex]
               ENDIF
               gAxisImpParamsValid = TRUE
